2. K-Means + EM / Gaussian Mixture

from sklearn.datasets import load_iris
from sklearn.cluster import KMeans
from sklearn.mixture import GaussianMixture

X = load_iris().data

# K-Means
kmeans = KMeans(n_clusters=3, random_state=42, n_init=10)
kmeans.fit(X)

# EM using K-Means centers
gmm = GaussianMixture(
    n_components=3,
    means_init=kmeans.cluster_centers_,
    random_state=42
)

gmm.fit(X)

print("K-Means Centers:")
print(kmeans.cluster_centers_)

print("\nCluster Probabilities:")
print(gmm.predict_proba(X)[:5])

print("\nFinal EM Clusters:")
print(gmm.predict(X))

3. K-Nearest Neighbors (KNN)

import pandas as pd
from sklearn.datasets import load_iris
from sklearn.model_selection import train_test_split
from sklearn.neighbors import KNeighborsClassifier

iris = load_iris()

X = iris.data
y = iris.target

X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.3, random_state=1
)

knn = KNeighborsClassifier(n_neighbors=10)
knn.fit(X_train, y_train)

print("Accuracy:", knn.score(X_test, y_test))
print("Prediction:", knn.predict([[6.5, 3.0, 5.8, 2.2]]))

4. Locally Weighted Regression

import numpy as np
from sklearn.model_selection import train_test_split

X = np.arange(1, 11).reshape(-1, 1)

y = np.array([
    2, 4, 5, 4, 5,
    7, 8, 9, 10, 12
])

X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.3, random_state=42
)

tau = 1

def predict(x):

    w = np.exp(
        -((X_train-x)**2)/(2*tau**2)
    ).ravel()

    A = np.c_[
        np.ones(len(X_train)),
        X_train
    ]

    theta = np.linalg.pinv(
        A.T @ np.diag(w) @ A
    ) @ A.T @ np.diag(w) @ y_train

    return np.array([1, x[0]]) @ theta


y_pred = np.array([
    predict(x) for x in X_test
])

print("Actual values:", y_test)
print("Predicted values:", y_pred)

6. Naive Bayes Classification

from sklearn.datasets import load_iris
from sklearn.model_selection import train_test_split
from sklearn.naive_bayes import GaussianNB
from sklearn.metrics import (
    accuracy_score,
    precision_score,
    recall_score,
    classification_report
)

iris = load_iris()

X_train, X_test, y_train, y_test = train_test_split(
    iris.data,
    iris.target,
    test_size=0.3,
    random_state=42
)

model = GaussianNB()

model.fit(X_train, y_train)

y_pred = model.predict(X_test)

print(
    "Accuracy:",
    accuracy_score(y_test, y_pred)
)

print(
    "Precision:",
    precision_score(
        y_test,
        y_pred,
        average='weighted'
    )
)

print(
    "Recall:",
    recall_score(
        y_test,
        y_pred,
        average='weighted'
    )
)

print(
    classification_report(
        y_test,
        y_pred,
        target_names=iris.target_names
    )
)
